Duplicate Prevention Algorithm & Scoring Formula

CivicTrack executes a weighted multi-factor validation algorithm to prevent duplicate municipal work orders:

Composite Score = (Category Affinity * 0.40) + (Ward Jurisdiction * 0.20) + (Spatial Proximity ≤ 150m * 0.25) + (Lexical Match * 0.15)
>> Threshold ≥ 0.55 triggers duplicate prevention radar, disallowing redundant ticket generation and appending community evidence.

AI Department Auto-Router (src/utils/aiRouter.ts) TypeScript 
export function determineDepartmentRouting(category: string, description: string) {
  const text = (category + ' ' + description).toLowerCase();

  if (text.includes('pothole') || text.includes('road') || text.includes('asphalt')) {
    return { deptId: 'DEPT-TRANS-01', department: 'Dept. of Transportation', slaHours: 24 };
  }
  if (text.includes('light') || text.includes('dark') || text.includes('lamp')) {
    return { deptId: 'DEPT-ELEC-02', department: 'Bureau of Street Lighting', slaHours: 24 };
  }
  if (text.includes('water') || text.includes('sewer') || text.includes('leak') || text.includes('pipe')) {
    return { deptId: 'DEPT-WATER-03', department: 'Water & Sewer Authority', slaHours: 12 };
  }
  if (text.includes('garbage') || text.includes('trash') || text.includes('waste')) {
    return { deptId: 'DEPT-WASTE-04', department: 'Public Sanitation', slaHours: 24 };
  }
  return { deptId: 'DEPT-GENERAL', department: 'Municipal Administration', slaHours: 48 };
}
Duplicate Prevention Distance Check (Haversine Formula) TypeScript 
export function calculateGeoDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371e3; // Earth radius in meters
  const phi1 = (lat1 * Math.PI) / 180;
  const phi2 = (lat2 * Math.PI) / 180;
  const deltaPhi = ((lat2 - lat1) * Math.PI) / 180;
  const deltaLambda = ((lon2 - lon1) * Math.PI) / 180;

  const a = Math.sin(deltaPhi / 2) ** 2 +
            Math.cos(phi1) * Math.cos(phi2) * Math.sin(deltaLambda / 2) ** 2;
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));

  return R * c; // Distance in meters
}
